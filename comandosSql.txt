CREATE DATABASE CAE;

USE CAE;

CREATE TABLE SOLICITUD (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100),
    apellidos VARCHAR(150),
    dni VARCHAR(9) UNIQUE,
    f_nac DATE,
    tlf VARCHAR(20),
    email VARCHAR(254),
    profesion VARCHAR(30),
    jornadaParcial TINYINT,
    idiomas VARCHAR(40)
);
