CREATE DATABASE CAE;

USE CAE

CREATE TABLE SOLICITUD (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100),
    apellidos VARCHAR(150),
    dni VARCHAR(9),
    f_nac DATE,
    tlf VARCHAR(20),
    email VARCHAR(150),
    profesion ENUM('soldadura', 'informatica', 'asistencia sociosanitaria'),
    jornadaParcial BOOLEAN,
    idiomas SET('Euskera', 'Español')
);
